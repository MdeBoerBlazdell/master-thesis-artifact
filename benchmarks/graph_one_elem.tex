\documentclass{article}

\usepackage{pgfplots}
\usepackage{pgfplotstable}
\usepackage{tikz}

\pgfplotsset{compat=1.18}
\pgfplotstableset{col sep=comma}
\pgfkeys{/pgf/number format/.cd,fixed,precision=0}

\begin{document}
\begin{figure*}[ht]
        \begin{tikzpicture}
            \begin{loglogaxis}[
                xlabel=N,
                ylabel=rows/s,
                anchor=west,
                legend style = {
                    at = {(0.0, -0.4)},
                    anchor = west,
                    legend columns = 2
                },
            ]

                \addplot+ [
                    error bars/.cd,
                        y explicit,
                        y dir=both,
                ] table [
                    x=N,
                    y=mean,
                    y error plus expr=\thisrow{max}-\thisrow{mean},
                    y error minus expr=\thisrow{mean}-\thisrow{min},
                ] {csvs/one_elem_selections.csv};

                \addplot+ [
                    error bars/.cd,
                        y explicit,
                        y dir=both,
                ] table [
                    x=N,
                    y=mean,
                    y error plus expr=\thisrow{max}-\thisrow{mean},
                    y error minus expr=\thisrow{mean}-\thisrow{min},
                ] {csvs/one_elem_slices.csv};

		\legend{Selections, Slices}		
            \end{loglogaxis}
        \end{tikzpicture}
\end{figure*}
\end{document}
